import java.util.*;
import java.io.*;

public class CollectionsAndIODemo {
    public static void main(String[] args) {
        // Collections Framework Demonstration
        Map<String, String> registry = new LinkedHashMap<>();
        registry.put("25EU02067", "Trivikram");
        registry.put("25EU02068", "Venkatesh");

        System.out.println("=== Student Lab Registry ===");
        registry.forEach((roll, name) -> System.out.println("Roll: " + roll + " | Student: " + name));

        // Stream I/O Operations
        String message = "Java Programming Lab 23AI5351 - Experiment 8 Completed.";
        try (ByteArrayInputStream in = new ByteArrayInputStream(message.getBytes());
             ByteArrayOutputStream out = new ByteArrayOutputStream()) {
            
            int c;
            while ((c = in.read()) != -1) {
                out.write(Character.toUpperCase(c));
            }
            System.out.println("\nProcessed Stream Output:");
            System.out.println(out.toString());
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
